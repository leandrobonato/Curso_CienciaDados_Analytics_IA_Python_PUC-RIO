CREATE TABLE IF NOT EXISTS CATALOGO_FONTES_DADOS(
  ID_FONTE INT NOT NULL,
  NOME_FONTE STRING,
  DESCRICAO_FONTE STRING,
  URL_FONTE STRING,
  DOCUMENTACAO_DOWNLOAD STRING,
  DATA_CARGA TIMESTAMP  
);

ALTER TABLE CATALOGO_FONTES_DADOS ADD CONSTRAINT PK_CATALOGO_FONTES_DADOS PRIMARY KEY (ID_FONTE);

CREATE TABLE IF NOT EXISTS CATALOGO_COLUNA_DADOS(
  ID_CAMPO INT NOT NULL,
  ID_FONTE INT NOT NULL,
  NOME_COLUNA STRING, 
  DESCRICAO_COLUNA STRING,
  TIPO_DADO STRING,
  CATEGORIA STRING,
  FORMATO_DADO STRING
);

ALTER TABLE catalogo_coluna_dados ADD CONSTRAINT PK_CATALOGO_COLUNA_DADOS PRIMARY KEY (ID_CAMPO);

ALTER TABLE CATALOGO_COLUNA_DADOS ADD CONSTRAINT FK_CATALOGO_COLUNA_DADOS FOREIGN KEY (ID_FONTE) REFERENCES CATALOGO_FONTES_DADOS(ID_FONTE);

CREATE TABLE IF NOT EXISTS CATALOGO_COLUNA_DADOS_OBS (
  ID_OBS INTEGER NOT NULL,
  ID_CAMPO INTEGER NOT NULL,
  OBSERVACAO STRING
);

ALTER TABLE catalogo_coluna_dados_obs ADD CONSTRAINT PK_CATALOGO_COLUNA_DADOS_OBS PRIMARY KEY (ID_OBS);
ALTER TABLE CATALOGO_COLUNA_DADOS_OBS ADD CONSTRAINT FK_CATALOGO_COLUNA_DADOS_OBS FOREIGN KEY (ID_CAMPO) REFERENCES CATALOGO_COLUNA_DADOS(ID_CAMPO);
 
CREATE TABLE IF NOT EXISTS CATALOGO_DADOS_REFERENCIA(
  ID_REFERENCIA INTEGER NOT NULL,
  ID_CAMPO INTEGER,
  NOME_REFERENCIA STRING,
  DESCRICAO_REFERENCIA_ENGLISH STRING,
  DESCRICAO_REFERENCIA_PORTUGUESE STRING
);

ALTER TABLE CATALOGO_DADOS_REFERENCIA ADD CONSTRAINT PK_CATALOGO_DADOS_REFERENCIA PRIMARY KEY (ID_REFERENCIA);

ALTER TABLE CATALOGO_DADOS_REFERENCIA ADD CONSTRAINT FK_CATALOGO_DADOS_REFERENCIA FOREIGN KEY (ID_CAMPO) REFERENCES CATALOG;
